import assert from 'node:assert/strict';
import { execFileSync } from 'node:child_process';
import { mkdir, writeFile } from 'node:fs/promises';
import { resolve, join } from 'node:path';
import { createFixture, fixtureProject } from '../../../../scripts/export-acceptance/fixture.ts';
import { render, withRenderSession } from '../../../../packages/renderer-browser/src/export/index.ts';
const out = resolve(process.argv[2]);
await mkdir(out);
await createFixture(out);
execFileSync('ffmpeg', ['-v', 'error', '-f', 'lavfi', '-i', 'testsrc2=s=960x540:r=25:d=1.2', '-c:v', 'libx264', '-pix_fmt', 'yuv420p', join(out, 'motion.mp4')]);
const loaded = await fixtureProject(out, 'quicksilver');
const project = { ...loaded, background: 'motion.mp4', css: loaded.css + '\n.theme-quicksilver .sc-bg-img {will-change:opacity !important;}' };
const report = { revision: execFileSync('git', ['rev-parse', 'HEAD'], { encoding: 'utf8' }).trim(), dirty: Boolean(execFileSync('git', ['status', '--porcelain', '--untracked-files=no'], { encoding: 'utf8' }).trim()), scope: 'Exact decoded equality for a looping 25 fps video across workers and repeated fresh pages. No tolerance.', status: 'running', results: [] };
try {
  await withRenderSession(async () => {
    for (let round = 0; round < 3; round++) for (const [width, height] of [[640, 360], [360, 640]]) {
      const observations = [];
      for (const concurrency of [1, 2, 4]) {
        const file = join(out, `${round}-${width}-${concurrency}.mp4`);
        const runtime = [];
        const result = await render({ ...project, width, height }, { projectDir: out, out: file, range: [1, 2.5], concurrency, onRuntime: event => runtime.push(event) });
        assert.equal(result.frames, 45);
        const hash = execFileSync('ffmpeg', ['-v', 'error', '-xerror', '-i', file, '-map', '0:v', '-f', 'hash', '-hash', 'sha256', '-'], { encoding: 'utf8' }).trim();
        observations.push({ concurrency, file, hash, runtime });
      }
      const equal = observations.every(item => item.hash === observations[0].hash);
      report.results.push({ round, width, height, equal, observations });
      await writeFile(join(out, 'report.json'), JSON.stringify(report, null, 2) + '\n');
      assert(equal, `Worker pixels differ: round ${round}, ${width}x${height}`);
      console.log(`round ${round}, ${width}x${height}: exact`);
    }
  });
  report.status = 'passed';
} catch (error) { report.status = 'failed'; report.error = String(error); throw error; }
finally { await writeFile(join(out, 'report.json'), JSON.stringify(report, null, 2) + '\n'); }
