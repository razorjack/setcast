#!/bin/bash
set -euo pipefail
root="/Users/razorjack/Projects/OpenSource/setcast/examples/demo/out/macos-followup-2026-11-03"
label="$1"
shift
test ! -e "$root/$label.txt"
printf '%q ' "$@" > "$root/$label.command.txt"
printf '\n' >> "$root/$label.command.txt"
date -u '+%Y-%m-%dT%H:%M:%SZ' > "$root/$label.start.txt"
status=0
"$@" > "$root/$label.txt" 2>&1 || status=$?
printf '%s\n' "$status" > "$root/$label.exit.txt"
cat "$root/$label.txt"
exit "$status"
