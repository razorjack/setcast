const out=resolve(process.argv[2]);await mkdir(out);
const source=resolve('packages/renderer-browser/src/export/video.tsx');
const original=await readFile(source,'utf8');
const before=`  const canvas = document.createElement('canvas');
  canvas.width = decoded.canvas.width;
  canvas.height = decoded.canvas.height;
  const context = canvas.getContext('2d')!;
  context.drawImage(decoded.canvas, 0, 0);`;
const after=`  const canvas = decoded.canvas;
  const context = canvas.getContext('2d') as CanvasRenderingContext2D | OffscreenCanvasRenderingContext2D;`;
assert.equal(original.split(before).length,2,'Unexpected video preparation source');
const sources={copy:original,direct:original.replace(before,after)};
const projectDir=resolve('examples/demo/out/replacement-corpus-v1/looping-video');
const loaded=await loadProject(projectDir,{themes,probeDuration:(audio,dir)=>probeAudio({audio},dir)});
const project=await new WholeSetReader().read(loaded.project,projectDir);
const control=new AbortController();for(const signal of ['SIGINT','SIGTERM'])process.once(signal,()=>control.abort(new Error(signal)));
const identity=await projectContent(project,projectDir,control.signal);
const report={revision:execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim(),identity,scope:'CORRECTNESS ONLY UNDER LOAD. Two seconds, one pair. Isolated video readback experiment. Each mode gets a fresh render session and exact source-file content recorded by SHA256. Original source restored in finally. Three alternating pairs of 1080p30 frames2700-2999, four workers. Rendering through cleanup; excludes project preparation and verification. Exact decoded video equality required; no Remotion claim.',status:'running',runs:[]};
const save=()=>writeFile(join(out,'report.json'),JSON.stringify(report,null,2)+'\n');
let reference;
try {
  for(let round=0;round<1;round++) for(const mode of round%2?['direct','copy']:['copy','direct']) {
    // Correctness-only run under observed load; no speed qualification.
    const observed=await projectContent(project,projectDir,control.signal);
    assert.equal(observed.projectDigest,identity.projectDigest);assert.equal(observed.assetsDigest,identity.assetsDigest);
    await writeFile(source,sources[mode]);
    const runtime=[],file=join(out,round+'-'+mode+'.mp4');
    const monitor=await measureProcesses(),load=measureLoad();let wallSeconds,activity,resources;const start=performance.now();
    try {
      await render(project,{...exportSettings(loaded.config.output),projectDir,out:file,range:[90,92],concurrency:4,signal:control.signal,onRuntime:event=>runtime.push(event)});
      wallSeconds=(performance.now()-start)/1000;
    } finally {activity=load.stop();resources=await monitor.stop();}
    const verification=await verifyVideo(file,{...project,frames:60},control.signal);
    reference??=verification.decodedVideoHash;
    const equal=verification.decodedVideoHash===reference;
    report.runs.push({round,mode,wallSeconds,equal,sourceHash:createHash('sha256').update(sources[mode]).digest('hex'),verification,runtime,activity,resources});await save();
    console.log(round,mode,wallSeconds,'equal',equal);
    assert(equal,'Direct decoder readback changed decoded output; reject without a speed claim.');
  }
  report.status='passed';

finally {await writeFile(source,original);await save();}
