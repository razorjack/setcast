#!/usr/bin/env python3
import concurrent.futures,json,subprocess
from pathlib import Path
ROOT=Path(__file__).resolve().parent
REPO=Path('/home/razorjack/Projects/opensource/setcast')
manifest=ROOT/'software-corpus/manifest.json'
cases=json.loads(manifest.read_text())['cases']
index={c['name']:str(ROOT/('font-quality-case-'+c['name'])/c['name']) for c in cases}
(ROOT/'font-quality-index.json').write_text(json.dumps(index,indent=2)+'\n')
def run(case):
    name=case['name'];label='font-quality-case-'+name
    result=subprocess.run(['bash',str(ROOT/'run.sh'),label,'node','scripts/corpus-quality.ts','--manifest',str(manifest),'--upstream','/tmp/setcast-linux-current-upstream','--case',name,'--out',str(ROOT/label)],cwd=REPO)
    return {'case':name,'exitCode':result.returncode}
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:results=list(pool.map(run,cases))
(ROOT/'font-quality-parallel-results.json').write_text(json.dumps(results,indent=2)+'\n')
assert all(r['exitCode']==0 for r in results),results
