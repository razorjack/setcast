#!/usr/bin/env python3
"""Report retained live observations without turning incomplete runs into passes."""
import json,statistics
from pathlib import Path
root=Path(__file__).resolve().parent
results=[]
for directory in sorted(root.glob('final-live-*')):
 path=directory/'report.json'
 if not path.is_file():continue
 report=json.loads(path.read_text())
 rows=[json.loads(line) for line in (directory/'samples.jsonl').read_text().splitlines()] if (directory/'samples.jsonl').exists() else []
 samples=[r for r in rows if r['kind']=='sample']
 def summary(rows):
  values={key:[r['details'][key] for r in rows if isinstance(r['details'].get(key),(int,float))] for key in ('heapBytes','nodeRssBytes','nodes','cached','renderFps','dropped')}
  return {key:{'min':min(v),'median':statistics.median(v),'max':max(v)} for key,v in values.items() if v}
 result={'name':directory.name,'report':str(path),'status':report['status'],'runtime':report['runtime'],'wallSeconds':report['wallSeconds'],'configSeconds':report['config']['seconds'],'samples':len(samples),'actions':report['actions'],'errors':report['errors'],'details':report['details'],'measurements':summary(samples),'quarterHourIntervals':[]}
 for index in sorted({int(r['wallSeconds']//900) for r in samples}):
  selected=[r for r in samples if int(r['wallSeconds']//900)==index]
  result['quarterHourIntervals'].append({'fromSeconds':index*900,'untilSeconds':(index+1)*900,'samples':len(selected),'measurements':summary(selected)})
 results.append(result)
output=root/'live-summary.json';output.write_text(json.dumps({'scope':'Live reports retain their original passed/failed/running status. Heap and runner RSS exclude GPU and browser native allocations. Quarter-hour intervals may include startup, restarts and partial final intervals; FPS is a measurement, not a universal pass threshold.','results':results},indent=2)+'\n')
for r in results:print(r['name'],r['status'],round(r['wallSeconds']),r['samples'],r['actions'],r['errors'])
