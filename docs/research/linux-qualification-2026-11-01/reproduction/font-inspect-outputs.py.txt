#!/usr/bin/env python3
import concurrent.futures,json,subprocess,sys
from pathlib import Path
ROOT=Path(__file__).resolve().parent
mode=sys.argv[1]
assert mode in ('calibration','tune','timing')
index=json.loads((ROOT/'font-quality-index.json').read_text())
comparisons=[Path(p)/'comparison/report.json' for p in index.values()] if mode=='calibration' else sorted(ROOT.glob(mode+'-*/report.json'))
jobs=[]
for path in comparisons:
    report=json.loads(path.read_text())
    for run in report['runs']:jobs.append((path,run))
def inspect(job):
    path,run=job
    output=path.parent/(Path(run['result']['file']).stem+'-blackframe.json')
    command=['ffmpeg','-hide_banner','-threads','2','-i',run['result']['file'],'-vf','blackframe=amount=99:threshold=20','-an','-f','null','-']
    result=subprocess.run(command,text=True,capture_output=True)
    lines=[line for line in result.stderr.splitlines() if 'pblack:' in line]
    report={'command':command,'exitCode':result.returncode,'file':run['result']['file'],'blackFrames':lines,'scope':'Every decoded frame scanned for >=99 percent pixels below level 20. This detects nearly black frames, not all missing-content defects.'}
    output.write_text(json.dumps(report,indent=2)+'\n')
    output.with_suffix('.txt').write_text(result.stdout+result.stderr)
    return {'report':str(output),'passed':result.returncode==0 and not lines}
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:results=list(pool.map(inspect,jobs))
(ROOT/(mode+'-blackframe-results.json')).write_text(json.dumps(results,indent=2)+'\n')
assert all(r['passed'] for r in results),results
if mode=='calibration':
    for name,directory in index.items():
        directory=Path(directory);reference=json.loads((directory/'reference/reference.json').read_text());offset=reference['frames']//2
        inputs=[directory/'reference/reference.mkv',directory/'comparison/0-browser.mp4',directory/'comparison/0-remotion.mp4']
        width=360 if reference['width']<reference['height'] else 640
        filters=[f'[{i}:v]select=eq(n\\,{offset}),scale={width}:-1[p{i}]' for i in range(3)]
        filters.append('[p0][p1][p2]hstack=inputs=3')
        command=['ffmpeg','-hide_banner','-v','error']
        for video in inputs:command+=['-threads','2','-i',str(video)]
        command+=['-filter_complex_threads','2','-filter_complex',';'.join(filters),'-frames:v','1','-update','1',str(directory/'inspection-midpoint.png')]
        if not (directory/'inspection-midpoint.png').exists():subprocess.run(command,check=True)
