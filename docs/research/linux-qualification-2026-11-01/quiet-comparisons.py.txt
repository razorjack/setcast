#!/usr/bin/env python3
"""Quiet worker screening and retained comparisons. Run quality separately afterward."""
import json, subprocess, sys
from pathlib import Path
ROOT = Path(__file__).resolve().parent
CHECKOUT = Path('/tmp/setcast-linux-current-upstream')
CASES = json.loads((ROOT/'software-corpus/manifest.json').read_text())['cases']
mode = sys.argv[1]
assert mode in ('tune', 'timing')
for case in CASES:
    choices = [('2','2'),('4','4'),('auto','auto')] if mode == 'tune' else [(None,None)]
    for browser,remotion in choices:
        label = f"{mode}-{case['name']}" + (f'-{browser}' if mode == 'tune' else '')
        if mode == 'timing':
            arms = {name: [] for name in ('browser','remotion')}
            for choice in ('2','4','auto'):
                report = json.loads((ROOT/f"tune-{case['name']}-{choice}/report.json").read_text())
                assert report['status'] == 'passed'
                for run in report['runs']:
                    arms[run['renderer']].append((run['wallSeconds'],choice))
            browser = min(arms['browser'])[1]
            remotion = min(arms['remotion'])[1]
            selection = {'scope':'One quiet screening pair per worker setting, followed by three independent alternating pairs. Screening variance is not characterized.', 'case':case['name'], 'candidates':arms, 'browser':browser, 'remotion':remotion}
            (ROOT/f"selection-{case['name']}.json").write_text(json.dumps(selection,indent=2)+'\n')
        command = ['bash',str(ROOT/'upstream-run.sh'),label,'node','scripts/render-compare.ts',case['directory'],'--at',str(case['at']),'--seconds',str(case['seconds']),'--repeat','1' if mode=='tune' else '3','--browser-concurrency',browser,'--remotion-concurrency',remotion,'--browser-crf','16','--browser-jpeg-quality','100','--remotion-crf','18','--idle-busy','.15','--out',str(ROOT/label)]
        subprocess.run(command,cwd=CHECKOUT,check=True)
